This is the January 2025 analysis file that produced the published tables, figure and incidence rate ratios (Swaminathan et al. 2025). It is kept for comparison: the differences page lists each place where the rebuilt scripts depart from it, and why.

NoteEdited for publication

For patient privacy, three lines were changed. The intervention date and the identifier of one corrected record are commented out and replaced by <removed for patient privacy>. A file location on the analyst’s computer is replaced by <data folder>/. The comment above the correction now gives its reason in general terms, “the recipient became ineligible before transplant”. The header was also edited: a neutral title, the author’s name spelled correctly, and the date of the file. Otherwise the file is unchanged. Because the removed values are needed, it does not run as shown.

It is shown below as it would read when knitted, with its prose formatted, but nothing in it is run. transplant_data_processed<-transplant_data_unprocessed%>%
  mutate(duration=as.numeric(End_Date-Date_of_consent))%>%
  uncount(duration)%>%
  group_by(Study_ID)%>%
  mutate(start_date=days(row_number())+Date_of_consent-days(1), stop_date=days(row_number())+Date_of_consent)%>%
  mutate(Post_Intervention=start_date>InterventionDate)%>%
  mutate(Is_Transplant_date=(Date_of_TXP==stop_date))%>%
  mutate(Is_Transplant_date=if_else(is.na(Is_Transplant_date), FALSE, Is_Transplant_date))%>%
  mutate(Transplant_Outcome=ifelse(Is_Transplant_date, Transplanted, 0))%>%
  mutate(HIV_Transplant_Outcome=ifelse(Is_Transplant_date, HIV_transplanted, 0))

names_Study_ID_crosswalk<-transplant_data_unprocessed%>%
  select(Study_ID, Candidate_Last_Name)

organ_offers_processed<-organ_offers_unprocessed%>%
  #Convert string to date format for Match_Date
  mutate(Match_Date=mdy(Match_Date))%>%
  #Remove duplicates
  distinct(PX_ID, Donor_ID, .keep_all=TRUE)%>%
  #Create variable that marks post-intervention period
  mutate(Post_Intervention=Match_Date>InterventionDate)%>%
  #Create variable that marks that organ was transplanted (regardless of where)
  mutate(organ_was_transplanted=case_when(
    Number_of_Kidneys_tranplanted_from_Donor>=1~TRUE,
    Number_of_Kidneys_tranplanted_from_Donor==0~FALSE,
    is.na(Number_of_Kidneys_tranplanted_from_Donor)~FALSE))%>%
  #Create variable that marks whether organ was declined for donor concerns potentially related to HIV status:
  mutate(Potential_HIV_Donor_decline=case_when(
           (Offer_Acceptance=="No") & (Primary_Refusal_Reason %in% HIV_reason_list)~TRUE,
           (Offer_Acceptance=="No") & !(Primary_Refusal_Reason %in% HIV_reason_list)~FALSE,
           !(Offer_Acceptance=="No")~NA
           ))%>%
  #Sort by Donor ID
  arrange(Donor_ID)%>%
  #Add Study_ID to dataset
  left_join(names_Study_ID_crosswalk)%>%
  #Fix one offer that was accepted but not transplanted: the recipient became ineligible before transplant
  # Removed for patient privacy: the record identifier of the corrected offer.
  # mutate(Offer_Acceptance=ifelse(Match_ID==<removed for patient privacy>, "Provisional Acceptance", Offer_Acceptance))

#Remove unneeded object

Demographics <- read_excel("<data folder>/Hope_Act_Excelsheet_9-18-2023.xlsx")%>%
  clean_names()%>%
  rename(recipient_age_yrs=age_yrs)%>%
  rename(recipient_sex=sex_10)%>%
  rename(recipient_race=race_11)%>%
  rename(recipient_ethnicity=ethnicity_12)%>%
  rename(donor_sex=sex_26)%>%
  rename(donor_race=race_27)%>%
  rename(donor_ethnicity=ethnicity_28)%>%
  mutate(PRA_1=str_first_number(class_i_pra_percent))%>%
  mutate(PRA_2=str_first_number(class_ii_pra_percent))%>%
  mutate(years_on_dialysis=str_first_number(time_on_dialysis))%>%
  mutate(months_on_dialysis=str_nth_number(time_on_dialysis, n = 2))%>%
  mutate(months_on_dialysis=ifelse(is.na(months_on_dialysis), 0, months_on_dialysis))%>%
  mutate(time_on_hd_years=years_on_dialysis+(1/12)*(months_on_dialysis))%>%
  mutate(donor_age_years=str_first_number(donor_age))%>%
  mutate(warm_ischemia_min=str_first_number(warm_ischemia_time))%>%
  mutate(cold_ischemia_hours=str_first_number(cold_ischemia_time))%>%
  mutate(cold_ischemia_min=str_nth_number(cold_ischemia_time, n = 2))%>%
  mutate(cold_ischemia_total=cold_ischemia_hours+(1/60)*cold_ischemia_min)%>%
  mutate(EndStatus=case_when(
    hiv_transplanted==1~"Transplanted from HIV D+ donor",
    hiv_transplanted==0 & transplanted==1~"Transplanted from HIV D- donor",
    TRUE~"Not transplanted"))%>%
  mutate(recipient_hcv_binary=ifelse(str_detect(recipient_hcv, "Negative"), "Negative", "Positive"))%>%
  mutate(PRA_2_binary=ifelse(PRA_2==0, "Zero", "Greather than zero"))
